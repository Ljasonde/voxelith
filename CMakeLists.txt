include(FetchContent)
function(dependency name path hash)
  set(url "https://codeload.github.com/${path}")
  if(EXISTS "${CMAKE_SOURCE_DIR}/build/downloads/${name}.zip")
    set(url "${CMAKE_SOURCE_DIR}/build/downloads/${name}.zip")
  endif()
  FetchContent_Declare(${name} URL "${url}" URL_HASH SHA256=${hash} SOURCE_SUBDIR unused)
  FetchContent_MakeAvailable(${name})
  set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
endfunction()

  get_filename_component(name "${src}" NAME)
  set(bin "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders/${name}.spv")
  add_custom_command(OUTPUT "${bin}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders"
    COMMAND Vulkan::glslc --target-env=vulkan1.3 -O "${src}" -o "${bin}" DEPENDS "${src}" ${includes} VERBATIM)
  list(APPEND spv "${bin}")
endforeach()
